package gatewayapp

import (
	"context"
	"os"
	"path/filepath"
	"testing"
	"time"

	"github.com/caelis-labs/caelis/control/agentbinding"
	"github.com/caelis-labs/caelis/control/memorytool"
)

// Run this identical test first with v0.5.2 (seed), then the release candidate
// (upgrade), using one isolated synthetic CAELIS_MEMORY_UPGRADE_DIR.
func TestMemoryReleasedStoreUpgrade(t *testing.T) {
	root := os.Getenv("CAELIS_MEMORY_UPGRADE_DIR")
	mode := os.Getenv("CAELIS_MEMORY_UPGRADE_MODE")
	if root == "" || (mode != "seed" && mode != "upgrade") {
		t.Fatal("explicit isolated directory and seed|upgrade mode required")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	workspace := filepath.Join(root, "workspace")
	if err := os.MkdirAll(workspace, 0700); err != nil {
		t.Fatal(err)
	}
	provider := newMemoryGoldenProvider(t)
	provider.EnableSteward()
	stack := newMemoryGoldenStack(t, provider, filepath.Join(root, "caelis"), workspace)
	defer stack.Close()
	if mode == "seed" {
		if defaultMemoryStewardProfile.Version != 1 {
			t.Fatal("seed must use released v0.5.2")
		}
		document, err := newAppConfigStore(filepath.Join(root, "caelis")).Load()
		if err != nil {
			t.Fatal(err)
		}
		if _, err := stack.testAgentBindings().BindAgentBinding(ctx, agentbinding.Binding{Handle: agentbinding.HandleSteward, ProfileID: document.ModelProfiles.DefaultProfileID, Effort: document.ModelProfiles.DefaultEffort}); err != nil {
			t.Fatal(err)
		}
	}
	waitMemoryGoldenCondition(t, ctx, "active Steward", func() bool { return stack.memorySteward != nil && stack.memorySteward.active.Load() })
	if mode == "seed" {
		// Stop the worker without disabling its persisted profile binding.
		if err := stack.memorySteward.wait(ctx); err != nil {
			t.Fatal(err)
		}
	} else {
		if defaultMemoryStewardProfile.Version != 2 {
			t.Fatal("upgrade must select profile 2")
		}
		waitMemoryGoldenCondition(t, ctx, "old queued work processed", func() bool {
			got, err := stack.memoryRuntime.Management().Inspect(ctx)
			return err == nil && got.Steward.CompletedJobs == 1 && got.Steward.ActiveRecords == 1
		})
	}
	configuration, err := stack.memoryRuntime.Management().GetStewardConfiguration(ctx)
	if err != nil {
		t.Fatal(err)
	}
	if len(configuration.Bindings) != 1 || configuration.Bindings[0].ProfileVersion != defaultMemoryStewardProfile.Version {
		t.Fatalf("binding=%+v", configuration.Bindings)
	}
	if mode == "upgrade" && len(configuration.Profiles) != 2 {
		t.Fatalf("historical profile missing: %+v", configuration.Profiles)
	}
	provider.Begin(memoryGoldenScenario{name: mode, actions: []memoryGoldenAction{{name: memorytool.RememberToolName, arguments: `{"text":"the release review language is Chinese"}`}}})
	active := runMemoryGoldenSession(t, ctx, stack, "session-upgrade-"+mode)
	assertMemoryGoldenResult(t, memoryGoldenToolResults(t, stack, active.SessionRef), memorytool.RememberToolName, `{"accepted":true}`)
	provider.AssertComplete(t)
	if mode == "seed" {
		got, err := stack.memoryRuntime.Management().Inspect(ctx)
		if err != nil || got.Steward.PendingJobs != 1 || provider.StewardCalls() != 0 {
			t.Fatalf("seed=%+v err=%v calls=%d", got, err, provider.StewardCalls())
		}
	} else {
		waitMemoryGoldenCondition(t, ctx, "new queued work processed", func() bool {
			got, err := stack.memoryRuntime.Management().Inspect(ctx)
			return err == nil && got.Steward.CompletedJobs == 2
		})
		if provider.StewardCalls() != 2 {
			t.Fatalf("model callbacks=%d want 2", provider.StewardCalls())
		}
	}
}
